ACTUAL=json.loads(CAPTURE.read_text())

class Result:
    def __init__(self,value):self.value=value
    def mappings(self):return self.value
    def scalar(self):return self.value
class Conn:
    dialect=dialect()
    def __init__(self,rows,report=True):self.rows=rows;self.report=report;self.calls=[]
    def execute(self,sql,args):
        self.calls.append((str(sql),args))
        if 'pg_get_viewdef' in str(sql):return Result(self.rows)
        if 'SELECT EXISTS' in str(sql):return Result(self.report)
        raise RuntimeError('Unexpected SQL')

def rows(kind):
    return [dict(relname=v['name'],reloptions=['security_barrier=true'],definition=v['definition'],source_owner=True,application_owner=True,
                 types=['integer','integer','text'],exact_return_rule=True,columns=['record_id','ingestion_id','authority_fingerprint']) for v in ACTUAL[kind]['views']]

def check(test,kind,mutate=None):
    values=rows(kind)
    if mutate:mutate(values)
    conn=Conn(values)
    if mutate:
        with test.assertRaises(rc.WorkerRoleError):rc._views(conn,'public')
    else:rc._views(conn,'public')
    test.assertTrue(all(call[1]['schema']=='public' for call in conn.calls))

class Review(unittest.TestCase):
    def test_actual_recorded_clean_definition_is_accepted(self):check(self,'clean')
    def test_actual_recorded_historical_definition_is_accepted(self):check(self,'historical')
    def test_pre708_without_report_schema_is_accepted(self):rc._views(Conn([],False),'public')
    def test_post708_missing_both_views_is_refused(self):
        with self.assertRaises(rc.WorkerRoleError):rc._views(Conn([],True),'public')
    def test_unrelated_version_of_either_locked_hash_never_accepted_under_snapshot(self):
        for kind in ['clean','historical']:
            with self.subTest(kind=kind):
                check(self,kind,lambda r:r[1].update(definition=r[0]['definition']))
    def test_original_constants_remain_exact(self):
        self.assertEqual(rc.REPORT_VIEW_SHA256,{'social_report_source_authority':'f140c09463d68b7878d4f7b199d04f15ef3b95e109cfbb5820572c9d5e0a84f0','social_report_source_authority_snapshot':'60d9f03c6a0586c236e51f25fba05bc3659c72101bc3021fb0f7b887152f78ed'})
        self.assertEqual(rc.REPORT_LOCKED_VIEW_HISTORICAL_SHA256,'a10869d91f9d43eb6713a33089592336a204650b25752c8c26065ec610c18de0')
        self.assertEqual(rc.READ_VIEWS,frozenset({'social_report_source_authority'}))
    def test_new_hash_only_affects_acceptance_and_does_not_mutate_catalog(self):
        conn=Conn(rows('historical'));rc._views(conn,'public')
        self.assertEqual(len(conn.calls),2)
        self.assertTrue(all(s.lstrip().startswith('SELECT') for s,p in conn.calls))
    def test_historical_fails_old_guard(self):
        source=subprocess.check_output(['git','show','HEAD:backend/social/worker/role_contract.py'],cwd=ROOT,text=True)
        original={};exec(compile(source,'<accepted-main-role-contract>','exec'),original)
        with self.assertRaises(original['WorkerRoleError']):original['_views'](Conn(rows('historical')),'public')
        original['_views'](Conn(rows('clean')),'public')
