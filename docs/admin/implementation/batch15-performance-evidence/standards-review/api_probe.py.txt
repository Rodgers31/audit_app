import runpy
import uvicorn
from pathlib import Path
from fastapi.testclient import TestClient
import requests
import httpx
fixture_path = '/Users/roger/.codex/worktrees/b15-performance/audit_app/frontend/browser-fixtures/reset-loading/api.py'
roots = []
def check(value, label):
    if not value:
        raise RuntimeError(label)
    print('PASS', label, flush=True)
def run(app, **kwargs):
    check(kwargs == {'host':'127.0.0.1','port':33518}, 'bind loopback33518')
    import legacy_browser_fixture_api as fixture
    roots.append(fixture.ROOT)
    check(fixture.ROOT.exists(), 'disposable root exists while fixture is running')
    check(str(fixture.database.engine.url).startswith('sqlite:///'), 'SQLite isolated')
    check(not fixture.main.AUTO_SEEDER_ENABLED and not fixture.main._WARMUP_ENABLED, 'startup mutators disabled')
    with TestClient(app) as client:
        for path in ('/auth/v1/user', '/rest/v1/profiles'):
            response=client.get(path, headers={'Origin':'http://127.0.0.1:33508'})
            check(response.status_code == 200 and response.json()['id'] == '00000000-0000-4000-8000-000000000001', path+' synthetic response')
            check(response.headers.get('access-control-allow-origin') == 'http://127.0.0.1:33508', path+' fixture CORS')
        response = client.get('/rest/v1/profiles', headers={'Origin':'https://example.invalid'})
        check(response.headers.get('access-control-allow-origin') is None, 'external Origin absent from CORS allowlist')
        response=client.get('/not-a-fixture-route')
        check(response.status_code == 404, 'absent path fails404')
    for send in (lambda: requests.get('https://example.invalid',timeout=.1), lambda: httpx.get('https://example.invalid',timeout=.1)):
        try:
            send()
        except (requests.ConnectionError, httpx.ConnectError) as e:
            check('External network disabled' in str(e), 'external request rejected before network')
        else:
            raise RuntimeError('external request allowed')
uvicorn.run=run
runpy.run_path(fixture_path,run_name='__main__')
check(len(roots)==1 and not roots[0].exists(), 'disposable DB root removed on shutdown')
