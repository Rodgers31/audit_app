BAD_VERSIONS=[None,True,False,0,-1,1.0,'1','01','1.0',[],{},float('nan'),float('inf')]
BAD_BOOLS=[None,0,1,-1,'true','false','0','1',[],{}]
BAD_HASHES=[None,True,0,'','a'*63,'A'*64,'z'*64,[],{}]

@pytest.mark.parametrize('field,invalid',[(f,v) for f,vs in [('expected_controls_version',BAD_VERSIONS),('publishing_enabled',BAD_BOOLS),('expected_state',BAD_HASHES),('reason',[None,True,0,'','  ',[],{},'a'*2001])] for v in vs])
def test_direct_and_http_strict_fields(media,field,invalid):
 svc,_=media;row=connected(svc.db);body=intent(svc.db,row).model_dump(mode='json');body[field]=invalid
 with pytest.raises((ValidationError,ValueError,TypeError)):
  set_account_publishing(SocialService(svc.db,available_adapters=REGISTRY),row.id,body)
 svc.db.rollback()
 if isinstance(invalid,float) and (invalid!=invalid or invalid==float('inf')):return
 client=mounted_client(media); response=client.patch(BASE+f'/accounts/{row.id}/publishing',json=body,headers=headers())
 assert response.status_code==422,response.text
 assert response.headers['cache-control']=='private, no-store'
 svc.db.expire_all();assert svc.db.get(SocialAccount,row.id).publishing_enabled is False
 assert not list(svc.db.scalars(select(SocialCommandReceipt)))

  raise HTTPException(denial)

@pytest.mark.parametrize('mutation',['external','scopes','capability','connection','page_version','parent_version','parent_revoked','page_expiry','parent_data','page_refresh','parent_refresh','hold','lease'])
def test_every_metadata_change_fences_old_intent(media_db,mutation):
 db=media_db;row=connected(db);body=intent(db,row);cred=db.get(SocialCredential,row.credential_id);parent=db.get(SocialCredential,cred.parent_credential_id);now=SocialService(db).now()
 if mutation=='external':row.external_account_id='replacement-page'
 elif mutation=='scopes':row.granted_scopes=['changed']
 elif mutation=='capability':row.capability_snapshot={}
 elif mutation=='connection':row.connection_state='disconnected'
 elif mutation=='page_version':cred.version+=1
 elif mutation=='parent_version':parent.version+=1
 elif mutation=='parent_revoked':parent.revoked_at=now
 elif mutation=='page_expiry':cred.access_expires_at=now+timedelta(days=1)
 elif mutation=='parent_data':parent.data_access_expires_at=now+timedelta(days=1)
 elif mutation=='page_refresh':cred.refresh_lease_token=uuid4();cred.refresh_lease_expires_at=now-timedelta(seconds=1)
 elif mutation=='parent_refresh':parent.refresh_lease_token=uuid4();parent.refresh_lease_expires_at=now-timedelta(seconds=1)
 elif mutation=='hold':row.hold_reason='outcome_unknown'
 elif mutation=='lease':retain_delivery(db,row)
 db.commit()
 with pytest.raises(SocialError,match='VERSION_CONFLICT'):execute(db,row,body)
 assert db.get(SocialAccount,row.id).publishing_enabled is False
 assert not list(db.scalars(select(SocialCommandReceipt)))

def test_suppressed_audit_cannot_certify_audited_command(media_db):
 db=media_db;row=connected(db);body=intent(db,row)
 db.execute(text("CREATE TRIGGER review_suppress_audit BEFORE INSERT ON social_audit_events WHEN NEW.action='account.publishing_updated' BEGIN SELECT RAISE(IGNORE); END"));db.commit()
 try:result=execute(db,row,body)
 except SocialError as error:
  assert error.code=='ACCOUNT_CONTROL_FAILED'
  db.rollback()
  assert db.get(SocialAccount,row.id).publishing_enabled is False
  assert not list(db.scalars(select(SocialCommandReceipt)))
  assert not list(db.scalars(select(SocialAuditEvent)))
  return
 assert len(list(db.scalars(select(SocialAuditEvent))))==1, f'Accepted account consent but zero audit events: {result}'
